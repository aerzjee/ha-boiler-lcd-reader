#include <tunables/global>

profile boiler_lcd_reader flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  capability net_bind_service,

  # Network access – required for Supervisor API, Ingress, and camera proxy
  network inet tcp,
  network inet udp,
  network inet6 tcp,
  network inet6 udp,
  network inet stream,
  network inet dgram,

  # Application directory – read-only
  /app/** r,
  /app/**.py r,

  # Python runtime
  /usr/bin/python3* rix,
  /usr/lib/python3/** r,
  /usr/local/lib/python3/** r,
  /usr/local/bin/** rix,

  # Persistent data directory – full read/write
  /data/** rw,
  /data/ rw,

  # Temporary files
  /tmp/** rw,
  /tmp/ rw,

  # Proc and sys access (required by numpy / OpenCV)
  /proc/sys/vm/mmap_min_addr r,
  /sys/devices/system/cpu/** r,
  @{PROC}/@{pid}/stat r,
  @{PROC}/@{pid}/maps r,
  @{PROC}/@{pid}/status r,
  @{PROC}/stat r,
  @{PROC}/meminfo r,
  @{PROC}/cpuinfo r,

  # Shared libraries
  /lib/** rm,
  /usr/lib/** rm,
  /usr/local/lib/** rm,

  # DNS resolution
  /etc/resolv.conf r,
  /etc/hosts r,
  /etc/nsswitch.conf r,
  /etc/host.conf r,
  /run/systemd/resolve/stub-resolv.conf r,
}
